🕷️ Props en Action

Les différentes instances du composant

💻 Comment les utiliser en React

// Importer le composant
import Spiderman from './Spiderman';

// Utiliser le composant avec différentes props
function App() {
  return (
    <>
      <Spiderman
        nom="Spider-Man"
        identité="Peter Parker"
        ennemi="Venom"
      />

      <Spiderman
        nom="Spider-Gwen"
        identité="Gwen Stacy"
        ennemi="Green Goblin"
      />

      <Spiderman
        nom="Miles Morales"
        identité="Miles Morales"
        ennemi="Kingpin"
      />
    </>
  );
}
