1️⃣ Sans Props (répétition)

// Composant unique
function App() {
  return (
    <>
      <Spiderman />
      <Spiderman />
      <Spiderman />
    </>
  );
}

// Problème:
// Tous les Spiderman
// sont identiques!

2️⃣ Avec Props (personnalisation)

// Composant avec props
function App() {
  return (
    <>
      <Spiderman
        nom="Spider-Man"
        identité="Peter"
        ennemi="Venom"
      />
      <Spiderman
        nom="Spider-Gwen"
        identité="Gwen"
        ennemi="Goblin"
      />
      <Spiderman
        nom="Miles"
        identité="Miles"
        ennemi="Kingpin"
      />
    </>
  );
}
